========================================
GIT - DAILY COMMANDS
========================================

# Check the current Git status
git status

# Add all changed/new files
git add .

# Commit the changes
git commit -m "Your commit message"

# Push commits to GitHub
git push origin main

# View commit history in compact format
git log --oneline

# Get the latest changes from GitHub
git pull


========================================
GIT - CREATE NEW REPOSITORY
========================================

# Initialize Git in the project folder
git init

# Create README.md
echo "# ShoppingCartPro-backend" >> README.md

# Add README.md to staging
git add README.md

# Create the first commit
git commit -m "first commit"

# Rename the current branch to main
git branch -M main

# Connect the local project to GitHub
git remote add origin https://github.com/davecoder95/ShoppingCartPro-backend.git

# Push the main branch to GitHub and set upstream
git push -u origin main

========================================
API - PRODUCT
========================================

# Get all products
GET http://localhost:8080/products

# Get product by ID
GET http://localhost:8080/products/1

# Create a product
POST http://localhost:8080/products

# Update a product
PUT http://localhost:8080/products/1

# Delete a product
DELETE http://localhost:8080/products/1
